
select kk.idkk, k.numerkontrakt
into #tblFaktOrder
FROM kontrakt_korekty kk
JOIN kontraktlista k   ON kk.id_kontrakt = k.idkontrakt
JOIN blokperiod bd     ON k.id_okres = bd.idblokperiod
                      AND bd.okresblokperiod = 24319
					  and (kk.datawystawieniafaktury between '2026-07-01' and '2026-07-31')
union
select kk.idkk, k.numerkontrakt
FROM kontrakt_korekty kk
JOIN kontraktlista k   ON kk.id_kontrakt = k.idkontrakt
JOIN blokperiod bd     ON k.id_okres = bd.idblokperiod
                      AND bd.okresblokperiod = 24319
					  and (kk.datawystawienia between '2026-07-01' and '2026-07-31')

CREATE INDEX ix_tmp_tblFaktOrder ON #tblFaktOrder (idkk)

CREATE INDEX ix_tmp2_tblFaktOrder ON #tblFaktOrder (numerkontrakt)

SELECT
    kk.IDKK,
    kk.ID_KONTRAKT,
    spot_agg.BLOKDATA_MAX,
    kse.ISPMP,
    ISNULL(luk1.cnt, 0)         AS iloscumow1,
    ISNULL(luk2.cnt, 0)         AS iloscumow2,
    ISNULL(fz_agg.cnt, 0)       AS IsFZ,
    fz_top.numerfaktury         AS numerFZ,
    fz_top.idFZ                 AS IdFZ,
    sz_top.NUMERZAMOWIENIA      AS numerZL,
    mop.IS_WO_DIGITAL,
    ISNULL(vm.iloscsub,  0) AS IsMasterZlecenie,
	ISNULL(vo.idkontrakt,  0) AS IsPodzlecenie,
	vo.numerkontrakt AS MasterNumer
	INTO #tblFaktDocs2
FROM kontrakt_korekty kk
JOIN #tblFaktOrder tb   ON kk.idkk = tb.idkk
-- MAX(BLOKDATA) zamiast subquery z 4 tabelami
LEFT JOIN (
    SELECT s.id_kontrakt, MAX(dt.BLOKDATA) AS BLOKDATA_MAX
    FROM SPOTLISTA s
    JOIN BLOKLISTA   bl ON s.ID_BLOK          = bl.IDBLOK
    JOIN BLOKPROGRAM bp ON bp.IDBLOKPROGRAM   = bl.ID_BLOKPROGRAM
    JOIN DNIOTWARTE  dt ON dt.IDDNIOTWARTE         = bp.ID_DNIOTWARTE
    GROUP BY s.id_kontrakt
) spot_agg ON spot_agg.id_kontrakt = kk.id_kontrakt
-- ISPMP
LEFT JOIN KONTRAKTLISTA_SYSTEM_EXT kse ON kse.NUMERKONTRAKT = tb.numerkontrakt
-- iloscumow1
LEFT JOIN (
    SELECT id_kontraktlista, COUNT(*) AS cnt
    FROM REZ_LU_KONTRAKT
    GROUP BY id_kontraktlista
) luk1 ON luk1.id_kontraktlista = kk.id_kontrakt
-- iloscumow2
LEFT JOIN (
    SELECT mo.id_multikontrakt, COUNT(*) AS cnt
    FROM multi_orders mo
    JOIN kontraktlista    kmaster ON kmaster.idkontrakt        = mo.id_masterkontrakt
    JOIN REZ_LU_KONTRAKT  luk     ON luk.id_kontraktlista      = kmaster.idkontrakt
    GROUP BY mo.id_multikontrakt
) luk2 ON luk2.id_multikontrakt = kk.id_kontrakt
-- IsFZ (COUNT)
LEFT JOIN (
    SELECT fzkk.id_kk, COUNT(*) AS cnt
    FROM fakturazbiorcza_kk fzkk
    JOIN fakturazbiorcza fz ON fzkk.id_fz = fz.idfz
    GROUP BY fzkk.id_kk
) fz_agg ON fz_agg.id_kk = kk.idkk
-- numerFZ + IdFZ (jeden JOIN zamiast dwóch subquery)
LEFT JOIN (
    SELECT fzkk.id_kk,
           fz.numerfaktury,
           fz.idFZ,
           ROW_NUMBER() OVER (PARTITION BY fzkk.id_kk ORDER BY fz.idfz) AS rn
    FROM fakturazbiorcza_kk fzkk
    JOIN fakturazbiorcza fz ON fzkk.id_fz = fz.idfz
) fz_top ON fz_top.id_kk = kk.idkk
        AND fz_top.rn = 1
-- numerZL (TOP 1 › ROW_NUMBER)
LEFT JOIN (
    SELECT szc.ID_SUBORDER,
           sz.NUMERZAMOWIENIA,
           ROW_NUMBER() OVER (PARTITION BY szc.ID_SUBORDER ORDER BY sz.IDSZ) AS rn
    FROM sponsoring_zamowienie_channels szc
    JOIN sponsoring_zamowienie sz ON sz.IDSZ = szc.ID_SZ
) sz_top ON sz_top.ID_SUBORDER = kk.ID_KONTRAKT
        AND sz_top.rn = 1
-- IS_WO_DIGITAL
LEFT JOIN MULTI_ORDERS_PARAMS mop ON mop.ID_KONTRAKT = kk.ID_KONTRAKT
LEFT JOIN vw_multiorder vm ON kk.ID_KONTRAKT = vm.idkontrakt
left join vw_masterorder vo ON kk.ID_KONTRAKT=vo.id_multikontrakt;
